## ReAct 是什么？

**Re**asoning + **Act**ing：

```
Thought: 用户想知道项目里有哪些文档，我应该先列出目录
Action: list_files(path="knowledge")
Observation: ["guide.md", "faq.md"]
Thought: 我可以读取 guide.md 来回答
Action: read_file(path="knowledge/guide.md")
Observation: （文件内容）
Answer: 根据文档，...
```

OpenAI Function Calling 把 Thought/Action 内化在模型推理中，我们负责执行 Action 并返回 Observation。

## 步骤 1：添加文件读取工具

创建 `agent/tools/file_reader.py`：

```python
# agent/tools/file_reader.py
from pathlib import Path

PROJECT_ROOT = Path(__file__).resolve().parent.parent.parent

READ_FILE_SCHEMA = {
    "type": "function",
    "function": {
        "name": "read_file",
        "description": "读取项目内指定相对路径的文本文件内容",
        "parameters": {
            "type": "object",
            "properties": {
                "path": {
                    "type": "string",
                    "description": "相对于项目根目录的文件路径，如 knowledge/guide.md",
                },
            },
            "required": ["path"],
        },
    },
}

LIST_FILES_SCHEMA = {
    "type": "function",
    "function": {
        "name": "list_files",
        "description": "列出指定目录下的文件（不含子目录内容详情）",
        "parameters": {
            "type": "object",
            "properties": {
                "path": {
                    "type": "string",
                    "description": "相对路径目录，如 knowledge 或 .",
                },
            },
            "required": ["path"],
        },
    },
}


def read_file(path: str) -> str:
    target = (PROJECT_ROOT / path).resolve()
    if not str(target).startswith(str(PROJECT_ROOT)):
        return "错误：不允许访问项目目录外的文件"
    if not target.exists():
        return f"错误：文件不存在 {path}"
    return target.read_text(encoding="utf-8")[:8000]  # 限制长度


def list_files(path: str = ".") -> str:
    target = (PROJECT_ROOT / path).resolve()
    if not str(target).startswith(str(PROJECT_ROOT)):
        return "错误：不允许访问项目目录外的路径"
    if not target.is_dir():
        return f"错误：目录不存在 {path}"
    files = [f.name for f in target.iterdir()]
    return ", ".join(files) if files else "（空目录）"
```

## 步骤 3：封装 ReAct Agent

创建 `agent/react_agent.py`：

```python
# agent/react_agent.py
import json
from rich.console import Console
from agent.memory import ConversationMemory
from agent.llm import client, MODEL
from agent.tools.registry import TOOL_SCHEMAS, execute_tool

console = Console()


class ReactAgent:
    def __init__(self, memory: ConversationMemory | None = None):
        self.memory = memory or ConversationMemory(
            system_prompt=(
                "你是一个能使用工具的 AI 助手。"
                "需要查时间、读文件、列目录时，请调用相应工具。"
                "请用中文回答。"
            )
        )

    def run(self, user_input: str, max_rounds: int = 8, verbose: bool = True) -> str:
        self.memory.add_user(user_input)

        for round_i in range(max_rounds):
            response = client.chat.completions.create(
                model=MODEL,
                messages=self.memory.to_api_messages(),
                tools=TOOL_SCHEMAS,
            )
            msg = response.choices[0].message

            if not msg.tool_calls:
                self.memory.add_assistant(msg.content)
                return msg.content

            # 记录 assistant 的 tool_calls
            self.memory.messages.append({
                "role": "assistant",
                "content": msg.content or "",
                "tool_calls": [
                    {
                        "id": tc.id,
                        "type": "function",
                        "function": {
                            "name": tc.function.name,
                            "arguments": tc.function.arguments,
                        },
                    }
                    for tc in msg.tool_calls
                ],
            })

            for tc in msg.tool_calls:
                args = json.loads(tc.function.arguments or "{}")
                if verbose:
                    console.print(f"[dim]🔧 工具: {tc.function.name}({args})[/]")
                try:
                    result = execute_tool(tc.function.name, args)
                except Exception as e:
                    result = f"工具执行失败: {e}"
                if verbose:
                    console.print(f"[dim]📋 结果: {result[:200]}...[/]" if len(result) > 200 else f"[dim]📋 结果: {result}[/]")

                self.memory.messages.append({
                    "role": "tool",
                    "tool_call_id": tc.id,
                    "content": result,
                })

        return "达到最大工具调用轮次，请简化问题后重试。"
```

## 常见问题

| 问题 | 解决 |
|------|------|
| 不读文件就瞎编 | system prompt 强调「必须先调工具」 |

| 工具调用链断裂 | 检查每轮 `tool` 消息是否带正确 `tool_call_id` |
