## 步骤 3：统一 main.py 入口

最终版 `main.py` 建议结构：

```python
"""Agent Demo CLI 入口。"""
import os
from rich.console import Console
from rich.markdown import Markdown

console = Console()
USE_LANGCHAIN = os.getenv("USE_LANGCHAIN", "1") == "1"


def run_chat():
    if USE_LANGCHAIN:
        from agent.langchain_agent import run_agent
        from agent.rag import build_vectorstore
        console.print("[dim]模式: LangChain Agent + RAG[/]")
        build_vectorstore()
        invoke = lambda text: run_agent(text)
    else:
        from agent.react_agent import ReactAgent
        agent = ReactAgent()
        console.print("[dim]模式: 手写 ReAct Agent[/]")
        invoke = lambda text: agent.run(text)

    console.print("[bold green]Agent Demo[/] | quit 退出 | /clear 清空 | /reindex 重建索引\n")

    while True:
        user_input = console.input("[bold cyan]你> [/]")
        cmd = user_input.strip().lower()

        if cmd in ("quit", "exit", "q"):
            break
        if user_input.strip() == "/clear":
            console.print("[yellow]请重启程序以清空 LangChain 历史，或自行扩展 memory[/]")
            continue
        if user_input.strip() == "/reindex":
            from agent.rag import build_vectorstore
            build_vectorstore(force_rebuild=True)
            console.print("[yellow]知识库索引已重建[/]\n")
            continue

        reply = invoke(user_input)
        console.print(Markdown(reply))
        console.print()


if __name__ == "__main__":
    run_chat()
```
